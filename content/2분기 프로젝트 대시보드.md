<script>
const sheetData = [
{ url: "https://docs.google.com/spreadsheets/d/e/2PACX-1vRMovf5Tjn4pSI5za_FGBogWnj8eXcEKSigPiHmiuLSX1wfEchn7w99Iwlrzd3taZuoQIV1Mf94bce2/pub?gid=0&single=true&output=csv", containerId: "table-1" },
{ url: "https://docs.google.com/spreadsheets/d/e/2PACX-1vRMovf5Tjn4pSI5za_FGBogWnj8eXcEKSigPiHmiuLSX1wfEchn7w99Iwlrzd3taZuoQIV1Mf94bce2/pub?gid=899664195&single=true&output=csv", containerId: "table-2" },
{ url: "https://docs.google.com/spreadsheets/d/e/2PACX-1vRMovf5Tjn4pSI5za_FGBogWnj8eXcEKSigPiHmiuLSX1wfEchn7w99Iwlrzd3taZuoQIV1Mf94bce2/pub?gid=1266255410&single=true&output=csv", containerId: "table-3" }
];
function loadAllData() {
sheetData.forEach(sheet => {
Papa.parse(sheet.url, {
download: true,
header: true,
complete: function(results) { drawTable(results.data, sheet.containerId); },
error: function(err) {
document.getElementById(sheet.containerId).innerHTML = "데이터 연동 실패";
console.error(err);
}
});
});
}
function drawTable(data, containerId) {
if (data.length === 0) {
document.getElementById(containerId).innerHTML = "데이터가 없습니다.";
return;
}
let tableHTML = '<table class="dashboard-table"><thead><tr>';
const columns = Object.keys(data[0]);
columns.forEach(col => { tableHTML += `<th>${col}</th>`; });
tableHTML += '</tr></thead><tbody>';
data.forEach(row => {
if (!row[columns[0]]) return; 
tableHTML += '<tr>';
columns.forEach(col => {
let cellValue = row[col] || '';
if (col.includes('진행') && cellValue) {
let percent = cellValue.replace(/[^0-9]/g, ''); 
if(percent) {
cellValue = `<div class="progress-bg"><div class="progress-fill" style="width: ${percent}%;"></div><div class="progress-text">${percent}%</div></div>`;
}
}
tableHTML += `<td>${cellValue}</td>`;
});
tableHTML += '</tr>';
});
tableHTML += '</tbody></table>';
document.getElementById(containerId).innerHTML = tableHTML;
}
setTimeout(loadAllData, 300); 
</script>
